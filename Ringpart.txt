-- Verificar si un jugador está dentro del ring
local function IsInRing(player)
    if not player or not player.Character then return false end
    
    local rootPart = player.Character:FindFirstChild("HumanoidRootPart")
    if not rootPart then return false end
    
    local pos = rootPart.Position
    return pos.X >= RingMin.X and pos.X <= RingMax.X and
           pos.Y >= RingMin.Y and pos.Y <= RingMax.Y and
           pos.Z >= RingMin.Z and pos.Z <= RingMax.Z
end

-- Verificar si un jugador está vivo
local function IsAlive(player)
    if not player or not player.Character then return false end
    
    local humanoid = player.Character:FindFirstChild("Humanoid")
    local rootPart = player.Character:FindFirstChild("HumanoidRootPart")
    
    if not humanoid or not rootPart then return false end
    if humanoid.Health <= 0 then return false end
    
    return true
end

-- Guardar los motores del personaje
local function SaveMotors(character)
    if not character then return nil end
    
    local motors = {}
    local descendants = character:GetDescendants()
    
    for _, obj in ipairs(descendants) do
        if obj:IsA("Motor6D") then
            motors[obj.Name] = {
                Part0 = obj.Part0,
                Part1 = obj.Part1,
                C0 = obj.C0,
                C1 = obj.C1
            }
        end
    end
    
    return motors
end

-- Restaurar los motores del personaje
local function RestoreMotors(motors)
    if not motors then return end
    
    for name, data in pairs(motors) do
        if data.Part0 and data.Part0.Parent then
            if data.Part0:FindFirstChild(name) == nil then
                local motor = Instance.new("Motor6D")
                motor.Name = name
                motor.Part0 = data.Part0
                motor.Part1 = data.Part1
                motor.C0 = data.C0
                motor.C1 = data.C1
                motor.Parent = data.Part0
            end
        end
    end
end

-- Verificar si el personaje está en estado de ragdoll
local function IsRagdolled(humanoid)
    if not humanoid then return false end
    
    local state = humanoid:GetState()
    return state == Enum.HumanoidStateType.Physics or
           state == Enum.HumanoidStateType.FallingDown or
           humanoid.PlatformStand or
           state == Enum.HumanoidStateType.Running
end

-- Anti-ragdoll permanente
local function FixRagdoll()
    if not AntiRagdoll_Active then return end
    
    local character = LocalPlayer.Character
    if not character then return end
    
    local humanoid = character:FindFirstChild("Humanoid")
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    
    if not humanoid or not rootPart then return end
    
    if IsRagdolled(humanoid) then
        local motors = SaveMotors(character)
        
        humanoid:ChangeState(Enum.HumanoidStateType.Running)
        humanoid.PlatformStand = false
        humanoid.Sit = false
        humanoid.AutoRotate = true
        
        rootPart.Velocity = Vector3.new(rootPart.Velocity.X, 20, rootPart.Velocity.Z)
        
        RestoreMotors(motors)
        
        print("🛡️ Anti-ragdoll activado - levantando personaje")
    end
end

-- Iniciar Anti-ragdoll
local function StartAntiRagdoll()
    AntiRagdoll_Active = true
    
    if AntiRagdollConnection then
        AntiRagdollConnection:Disconnect()
    end
    
    AntiRagdollConnection = RunService.Heartbeat:Connect(FixRagdoll)
    print("🛡️ Anti-ragdoll permanente activado")
end

local function CreateESPForPlayer(player)
    if not player or player == LocalPlayer then return end
    if not ESP_Enabled then return end
    
    -- Crear folder para el ESP
    if not ESP_Folder or not ESP_Folder.Parent then
        ESP_Folder = Instance.new("Folder")
        ESP_Folder.Name = "ESP_" .. tostring(tick())
        ESP_Folder.Parent = game:GetService("CoreGui")
    end
    
    -- Eliminar ESP existente
    local existing = ESP_Folder:FindFirstChild(player.Name)
    if existing then
        existing:ClearAllChildren()
        existing:Destroy()
    end
    
    local playerFolder = Instance.new("Folder")
    playerFolder.Name = player.Name
    playerFolder.Parent = ESP_Folder
    
    ESP_Objects[player] = {}
    
    local function CreateESPForCharacter(character)
        if not character or not ESP_Enabled then return end
        
        task.spawn(function()
            local rootPart = character:WaitForChild("HumanoidRootPart", 2)
            if not rootPart then return end
            
            -- Caja alrededor del jugador
            local box = Instance.new("BoxHandleAdornment")
            box.Name = "Box"
            box.Size = Vector3.new(4, 7, 4)
            box.Transparency = 0.7
            box.ZIndex = 0
            box.AlwaysOnTop = true
            box.Visible = true
            box.Adornee = rootPart
            box.Parent = playerFolder
            
            -- Nombre del jugador
            local billboard = Instance.new("BillboardGui")
            billboard.Name = "NameTag"
            billboard.Enabled = true
            billboard.Size = UDim2.new(0, 200, 0, 50)
            billboard.AlwaysOnTop = true
            billboard.StudsOffset = Vector3.new(0, 5, 0)
            billboard.MaxDistance = 1000
            billboard.Adornee = rootPart
            billboard.Parent = playerFolder
            
            local nameLabel = Instance.new("TextLabel")
            nameLabel.Parent = billboard
            nameLabel.Name = "Tag"
            nameLabel.BackgroundTransparency = 1
            nameLabel.Position = UDim2.new(0, -50, 0, 0)
            nameLabel.Size = UDim2.new(0, 300, 0, 20)
            nameLabel.TextSize = 22
            nameLabel.TextStrokeColor3 = Color3.new(0, 0, 0)
            nameLabel.TextStrokeTransparency = 0.4
            nameLabel.Text = player.Name
            nameLabel.Font = Enum.Font.SourceSansBold
            nameLabel.TextScaled = false
            
            -- Highlight
            local highlight = Instance.new("Highlight")
            highlight.Name = "Highlight"
            highlight.Adornee = character
            highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            highlight.FillTransparency = 0.5
            highlight.OutlineTransparency = 0
            highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
            highlight.Parent = playerFolder
            
            -- Colores según equipo
            local color = player.TeamColor and player.TeamColor.Color or Color3.fromRGB(255, 255, 255)
            box.Color3 = color
            nameLabel.TextColor3 = color
            highlight.FillColor = color
        end)
    end
    
    -- Crear ESP para el personaje actual
    if player.Character then
        CreateESPForCharacter(player.Character)
    end
    
    -- Conectar a CharacterAdded
    local connection = player.CharacterAdded:Connect(function(character)
        task.wait(0.5)
        if ESP_Enabled and playerFolder.Parent then
            CreateESPForCharacter(character)
        end
    end)
    
    ESP_Objects[player].charConnection = connection
    ESP_Objects[player].folder = playerFolder
end

local function RemoveESPForPlayer(player)
    local data = ESP_Objects[player]
    if data then
        if data.charConnection then
            data.charConnection:Disconnect()
        end
        ESP_Objects[player] = nil
    end
    
    if ESP_Folder and ESP_Folder.Parent then
        local folder = ESP_Folder:FindFirstChild(player.Name)
        if folder then
            folder:ClearAllChildren()
            folder:Destroy()
        end
    end
end

local function ToggleESP()
    ESP_Enabled = not ESP_Enabled
    
    if ESP_Enabled then
        print("🔍 Activando ESP...")
        
        if ESP_Folder and ESP_Folder.Parent then
            ESP_Folder:Destroy()
        end
        
        ESP_Folder = Instance.new("Folder")
        ESP_Folder.Name = "ESP_" .. tostring(tick())
        ESP_Folder.Parent = game:GetService("CoreGui")
        
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                CreateESPForPlayer(player)
            end
        end
        
        if PlayerAddedConnection then
            PlayerAddedConnection:Disconnect()
        end
        
        PlayerAddedConnection = Players.PlayerAdded:Connect(function(player)
            if player ~= LocalPlayer then
                CreateESPForPlayer(player)
            end
        end)
        
        if PlayerRemovingConnection then
            PlayerRemovingConnection:Disconnect()
        end
        
        PlayerRemovingConnection = Players.PlayerRemoving:Connect(function(player)
            RemoveESPForPlayer(player)
        end)
        
        print("✅ ESP Activado")
    else
        print("🔍 Desactivando ESP...")
        
        if PlayerAddedConnection then
            PlayerAddedConnection:Disconnect()
            PlayerAddedConnection = nil
        end
        
        if PlayerRemovingConnection then
            PlayerRemovingConnection:Disconnect()
            PlayerRemovingConnection = nil
        end
        
        for player, data in pairs(ESP_Objects) do
            if data.charConnection then
                data.charConnection:Disconnect()
            end
        end
        ESP_Objects = {}
        
        if ESP_Folder then
            ESP_Folder:ClearAllChildren()
            ESP_Folder:Destroy()
            ESP_Folder = nil
        end
        
        print("✅ ESP Desactivado")
    end
end

local function StopChase()
    if ChaseConnection then
        ChaseConnection:Disconnect()
        ChaseConnection = nil
    end
end

local function StartChase(target)
    if ChaseConnection then
        StopChase()
    end
    
    if not target or not IsAlive(target) then return end
    
    local function ChaseLoop()
        if not Chase_Enabled then
            StopChase()
            return
        end
        
        local character = LocalPlayer.Character
        if not character then
            StopChase()
            return
        end
        
        local rootPart = character:FindFirstChild("HumanoidRootPart")
        if not rootPart then return end
        
        local targetChar = target.Character
        if not targetChar or not IsAlive(target) then
            StopChase()
            return
        end
        
        local targetRoot = targetChar:FindFirstChild("HumanoidRootPart")
        if not targetRoot then return end
        
        local targetPos = Vector3.new(targetRoot.Position.X, Config.ChaseHeight, targetRoot.Position.Z)
        
        local distance = (targetPos - rootPart.Position).Magnitude
        
        if distance < 3 then
            pcall(function()
                rootPart.CFrame = CFrame.new(targetPos, targetRoot.Position)
                rootPart.Velocity = Vector3.new(0, 0, 0)
                rootPart.RotVelocity = Vector3.new(0, 0, 0)
            end)
        else
            local speed = Config.ChaseSpeed
            local newPos = rootPart.Position + (targetPos - rootPart.Position).Unit * speed * (task.wait() or 0.016)
            
            newPos = Vector3.new(
                math.clamp(newPos.X, RingMin.X, RingMax.X),
                Config.ChaseHeight,
                math.clamp(newPos.Z, RingMin.Z, RingMax.Z)
            )
            
            pcall(function()
                rootPart.CFrame = CFrame.new(newPos, targetRoot.Position)
                rootPart.Velocity = Vector3.new(0, 0, 0)
                rootPart.RotVelocity = Vector3.new(0, 0, 0)
            end)
        end
    end
    
    ChaseConnection = RunService.Heartbeat:Connect(ChaseLoop)
end

local function ToggleChase()
    Chase_Enabled = not Chase_Enabled
    
    if Chase_Enabled then
        print("🎯 PERSECUCIÓN ACTIVADA")
        
        local character = LocalPlayer.Character
        if character then
            local rootPart = character:FindFirstChild("HumanoidRootPart")
            if rootPart then
                pcall(function()
                    rootPart.CFrame = CFrame.new(Vector3.new(-1896, 24, -70))
                    rootPart.Velocity = Vector3.new(0, 0, 0)
                    rootPart.RotVelocity = Vector3.new(0, 0, 0)
                end)
            end
        end
        
        -- Loop de persecución
        if ChaseLoopThread then
            task.cancel(ChaseLoopThread)
        end
        
        ChaseLoopThread = task.spawn(function()
            while Chase_Enabled do
                task.wait(0.1)
                
                if TargetPlayer and IsInRing(TargetPlayer) and IsAlive(TargetPlayer) then
                    StartChase(TargetPlayer)
                    
                    if FakeInvis_Active then
                        ToggleFakeInvis()
                    end
                else
                    StopChase()
                    
                    if Chase_Enabled and not FakeInvis_Active then
                        ToggleFakeInvis()
                    end
                end
            end
        end)
        
        print("✅ Persecución activada - Altura fija: " .. Config.ChaseHeight)
    else
        print("🎯 PERSECUCIÓN DESACTIVADA")
        
        StopChase()
        
        if ChaseLoopThread then
            task.cancel(ChaseLoopThread)
            ChaseLoopThread = nil
        end
        
        if FakeInvis_Active then
            ToggleFakeInvis()
        end
        
        print("✅ Persecución desactivada")
    end
end

local InvisConnections = {}
local OriginalCFrame = nil

local function CleanupInvisConnections()
    for _, data in ipairs(InvisConnections) do
        if data.type == "step" then
            pcall(function()
                RunService:UnbindFromRenderStep(data.step)
            end)
        elseif data.type == "event" then
            pcall(function()
                data.conn:Disconnect()
            end)
        end
    end
    InvisConnections = {}
end

local function ToggleFakeInvis()
    FakeInvis_Active = not FakeInvis_Active
    
    if FakeInvis_Active then
        print("👻 Invisibilidad falsa ACTIVADA")
        
        local character = LocalPlayer.Character
        if not character then
            print("⚠️ No hay personaje para activar invisibilidad")
            FakeInvis_Active = false
            return
        end
        
        local rootPart = character:FindFirstChild("HumanoidRootPart")
        if not rootPart then
            print("⚠️ No se encontró HumanoidRootPart")
            FakeInvis_Active = false
            return
        end
        
        OriginalCFrame = rootPart.CFrame
        
        CleanupInvisConnections()
        
        local stepName1 = "FakeStep_" .. tostring(tick()) .. "_" .. math.random(1000, 9999)
        local stepName2 = "FakeStep_" .. tostring(tick()) .. "_" .. math.random(1000, 9999)
        
        RunService:BindToRenderStep(stepName1, Enum.RenderPriority.Camera.Value - 5, function()
            if rootPart and FakeInvis_Active and rootPart.Parent then
                rootPart.CFrame = OriginalCFrame
            end
        end)
        table.insert(InvisConnections, {step = stepName1, type = "step"})
        
        RunService:BindToRenderStep(stepName2, Enum.RenderPriority.Camera.Value + 5, function()
            if rootPart and FakeInvis_Active and rootPart.Parent then
                OriginalCFrame = rootPart.CFrame
                rootPart.CFrame = CFrame.new(rootPart.Position + Vector3.new(0, -Config.FakeInvisOffset, 0)) * CFrame.Angles(0, 0, math.rad(180))
            end
        end)
        table.insert(InvisConnections, {step = stepName2, type = "step"})
        
        local conn1 = RunService.PreAnimation:Connect(function()
            if rootPart and FakeInvis_Active and rootPart.Parent then
                rootPart.CFrame = OriginalCFrame
            end
        end)
        table.insert(InvisConnections, {conn = conn1, type = "event"})
        
        local conn2 = RunService.PostSimulation:Connect(function()
            if rootPart and FakeInvis_Active and rootPart.Parent then
                OriginalCFrame = rootPart.CFrame
                rootPart.CFrame = CFrame.new(rootPart.Position + Vector3.new(0, -Config.FakeInvisOffset, 0)) * CFrame.Angles(0, 0, math.rad(180))
            end
        end)
        table.insert(InvisConnections, {conn = conn2, type = "event"})
        
        print("✅ Invisibilidad activada - 6000 studs abajo")
    else
        print("👁️ Invisibilidad falsa DESACTIVADA")
        
        CleanupInvisConnections()
        
        local character = LocalPlayer.Character
        if character then
            local rootPart = character:FindFirstChild("HumanoidRootPart")
            if rootPart and OriginalCFrame then
                pcall(function()
                    rootPart.CFrame = OriginalCFrame
                end)
            end
        end
        
        print("✅ Invisibilidad desactivada")
    end
end

local function FindPlayerByName(name)
    if name == "" or name:lower() == "todos" or name:lower() == "all" then
        return nil, "TODOS"
    end
    
    local search = name:lower():gsub("%s+", "")
    
    -- Buscar coincidencia exacta
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            if player.Name:lower() == search or player.DisplayName:lower() == search then
                return player, player.Name
            end
        end
    end
    
    -- Buscar por coincidencia parcial (mínimo 3 letras)
    if #search >= 3 then
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                local nameLower = player.Name:lower()
                local displayLower = player.DisplayName:lower()
                
                if nameLower:sub(1, #search) == search or displayLower:sub(1, #search) == search then
                    return player, player.Name
                end
            end
        end
        
        -- Buscar contiene el texto
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                local nameLower = player.Name:lower()
                local displayLower = player.DisplayName:lower()
                
                if nameLower:find(search, 1, true) or displayLower:find(search, 1, true) then
                    return player, player.Name
                end
            end
        end
    end
    
    if #search > 0 and #search < 3 then
        return false, "Mínimo 3 letras"
    end
    
    return false, "No encontrado"
end

local function CreateRing()
    print("⚡ Creando cubo del ring")
    
    local ring = Instance.new("Part")
    ring.Name = "CUBO_INVISIBLE_" .. LocalPlayer.Name
    ring.Size = Config.RingSize
    ring.Position = Config.RingPos
    ring.Anchored = true
    ring.CanCollide = false
    ring.Transparency = Config.RingTransparency
    ring.BrickColor = BrickColor.new(Config.RingColor)
    ring.Material = Enum.Material.SmoothPlastic
    ring.Shape = Enum.PartType.Block
    ring.CastShadow = true
    ring.Parent = Workspace
    
    -- Crear bordes
    local function CreateEdge(size, position)
        local edge = Instance.new("Part")
        edge.Size = size
        edge.Position = position
        edge.Anchored = true
        edge.CanCollide = false
        edge.Transparency = 0.7
        edge.BrickColor = BrickColor.new(Config.RingColor)
        edge.Material = Enum.Material.Neon
        edge.Parent = ring
        return edge
    end
    
    local pos = Config.RingPos
    local size = Config.RingSize
    
    -- Bordes horizontales superiores
    CreateEdge(Vector3.new(size.X, 0.2, 0.2), pos + Vector3.new(0, size.Y/2, size.Z/2))
    CreateEdge(Vector3.new(size.X, 0.2, 0.2), pos + Vector3.new(0, size.Y/2, -size.Z/2))
    CreateEdge(Vector3.new(0.2, 0.2, size.Z), pos + Vector3.new(size.X/2, size.Y/2, 0))
    CreateEdge(Vector3.new(0.2, 0.2, size.Z), pos + Vector3.new(-size.X/2, size.Y/2, 0))
    
    -- Bordes horizontales inferiores
    CreateEdge(Vector3.new(size.X, 0.2, 0.2), pos + Vector3.new(0, -size.Y/2, size.Z/2))
    CreateEdge(Vector3.new(size.X, 0.2, 0.2), pos + Vector3.new(0, -size.Y/2, -size.Z/2))
    CreateEdge(Vector3.new(0.2, 0.2, size.Z), pos + Vector3.new(size.X/2, -size.Y/2, 0))
    CreateEdge(Vector3.new(0.2, 0.2, size.Z), pos + Vector3.new(-size.X/2, -size.Y/2, 0))
    
    -- Bordes verticales
    CreateEdge(Vector3.new(0.2, size.Y, 0.2), pos + Vector3.new(size.X/2, 0, size.Z/2))
    CreateEdge(Vector3.new(0.2, size.Y, 0.2), pos + Vector3.new(size.X/2, 0, -size.Z/2))
    CreateEdge(Vector3.new(0.2, size.Y, 0.2), pos + Vector3.new(-size.X/2, 0, size.Z/2))
    CreateEdge(Vector3.new(0.2, size.Y, 0.2), pos + Vector3.new(-size.X/2, 0, -size.Z/2))
    
    print("✅ Cubo del ring creado")
    return ring
end

            TargetPlayer = result
            TargetName = name

            helpMessage.Text = "✓ Encontrado: " .. name

            end

            if helpMessage.Text:sub(1, 1) == "✓" or

            end
        end

            UpdateESPButton()
        elseif input.KeyCode == Enum.KeyCode.F then
            ToggleChase()
            UpdateChaseButton()
        elseif input.KeyCode == Enum.KeyCode.T then
            ToggleFakeInvis()
            UpdateInvisButton()
        end
    end)

        end
    end)
    
    -- Cuando un jugador se va
    if PlayerRemovingConnection then
        PlayerRemovingConnection:Disconnect()
    end
    
    PlayerRemovingConnection = Players.PlayerRemoving:Connect(function(player)
        if TargetPlayer == player then
            TargetPlayer = nil
            TargetName = "TODOS"
            targetBox.Text = ""
            UpdateTargetStatus()
            helpMessage.Text = "⚠️ El objetivo salió"
            helpMessage.TextColor3 = Color3.fromRGB(255, 150, 50)
            
            if Chase_Enabled then
                StopChase()
            end
        end
        
        if ESP_Enabled then
            RemoveESPForPlayer(player)
        end
    end)
    
    -- Inicializar botones
    UpdateESPButton()
    UpdateChaseButton()
    UpdateInvisButton()
    UpdateTargetStatus()
    
    return mainFrame
